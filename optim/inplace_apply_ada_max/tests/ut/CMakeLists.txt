set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# CANN_HOME resolution (same logic as tests/tiling/CMakeLists.txt).
if(NOT DEFINED CANN_HOME)
    if(DEFINED ENV{ASCEND_HOME_PATH})
        set(CANN_HOME $ENV{ASCEND_HOME_PATH})
    elseif(DEFINED ENV{ASCEND_TOOLKIT_HOME})
        set(CANN_HOME $ENV{ASCEND_TOOLKIT_HOME})

    endif()

if(NOT DEFINED ASCEND_HOST_ARCH)
    execute_process(COMMAND uname -m
        OUTPUT_VARIABLE _UNAME_M OUTPUT_STRIP_TRAILING_WHITESPACE)
    set(ASCEND_HOST_ARCH ${_UNAME_M})

set(ASCEND_HOST_LINUX "${ASCEND_HOST_ARCH}-linux")

set(CANN_INC ${CANN_HOME}/${ASCEND_HOST_LINUX}/include)

# PKG_DIR: operator package root (two levels up: tests/ut -> tests -> package).
set(PKG_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../..)

# UT executable name (glob-friendly for the skill's verification gate #3).
set(UT_EXE inplace_apply_ada_max_tiling_ut)

add_executable(${UT_EXE}
    test_inplace_apply_ada_max_tiling.cpp
    test_inplace_apply_ada_max_branch0_tiling.cpp
    test_inplace_apply_ada_max_branch1_tiling.cpp
    test_inplace_apply_ada_max_branch2_tiling.cpp
    test_inplace_apply_ada_max_branch3_tiling.cpp
    ${PKG_DIR}/op_host/arch35/inplace_apply_ada_max_tiling_host_arch35.cpp
)
target_include_directories(${UT_EXE} PRIVATE
    ${CANN_INC}                       # graph/types.h (ge::DataType)
    ${PKG_DIR}/op_host/arch35         # inplace_apply_ada_max_tiling_host_arch35.h
    ${PKG_DIR}/op_kernel/arch35       # inplace_apply_ada_max_tiling_data.h
)
# NOTE: do NOT define _GLIBCXX_USE_CXX11_ABI=0 here. The system gtest
# (/usr/lib64/libgtest.so) is built with the new C++11 ABI; the UT must match.
# This is safe because the UT only uses ge::DataType (an enum) and the pure
# tiling functions (int64_t/bool only) — no CANN std::string ABI interaction.
target_compile_options(${UT_EXE} PRIVATE
    -Wall -Wno-deprecated-declarations -Wno-unused-variable
)

# Link system gtest (available at /usr/lib64/libgtest.so on this env).
find_package(Threads REQUIRED)
target_link_directories(${UT_EXE} PRIVATE ${CANN_LIB})
target_link_libraries(${UT_EXE} PRIVATE
    gtest
    gtest_main
    Threads::Threads
)

# Hook the UT into the `binary` target so `make binary package` (build.sh)
# compiles it. The skill's verification gate #3 requires the UT executable to
# exist after build.sh.
if(TARGET binary)
    add_dependencies(binary ${UT_EXE})
endif()

# Place the built UT executable under tests/ut/ (next to the package build dir)
# so the skill's verification gate can find it via a glob.
set_target_properties(${UT_EXE} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
)
